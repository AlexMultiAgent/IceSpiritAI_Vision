```kotlin
// app/src/main/java/com/icespiritai/offline/updater/ApkSignatureVerifier.kt
package com.icespiritai.offline.updater

import android.util.Log
import java.io.File
import java.security.MessageDigest
import java.util.jar.JarFile

/**
 * Reads the SHA-256 fingerprint of the first v1 signer certificate of an
 * APK file. The v1 signature block (META-INF/CERT.{RSA,DSA,EC}) is the
 * only signing scheme whose contents are readable via plain [JarFile] —
 * v2/v3 live inside the APK Signing Block which `java.util.jar` cannot
 * extract.
 *
 * Returns lowercase hex (64 chars) on success, `null` if the file is not
 * a JAR/APK, lacks a v1 manifest, or its certificate cannot be parsed.
 *
 * MUST stay aligned with `app/build.gradle.kts:extractApkCertificateSha256`
 * (build-time mirror) and `AppVersionInfo.signerCertSha256` (JSON contract).
 */
object ApkSignatureVerifier {

    private const val TAG = "ApkSignatureVerifier"

    private val CERT_ENTRIES = listOf("META-INF/CERT.RSA", "META-INF/CERT.DSA", "META-INF/CERT.EC")

    fun readFirstSignerCert(apk: File): String? = try {
        JarFile(apk).use { jar ->
            for (entry in CERT_ENTRIES) {
                val e = jar.getEntry(entry) ?: continue
                jar.getInputStream(e).use { stream ->
                    val bytes = stream.readBytes()
                    val certBytes = skipAsn1Envelope(bytes)
                    if (certBytes != null) {
                        val md = MessageDigest.getInstance("SHA-256")
                        return@use md.digest(certBytes).toLowerHex()
                    }
                }
            }
            null
        }
    } catch (t: Throwable) {
        Log.w(TAG, "readFirstSignerCert failed: ${t.javaClass.simpleName}")
        null
    }

    /**
     * The CERT.RSA file is a PKCS#7 SignedData structure (ASN.1 SEQUENCE).
     * The first certificate is wrapped in it. We extract just enough DER to
     * feed SHA-256: the simplest correct path is to hash the entire PKCS#7
     * blob (since the SHA-256 of the cert matches only if both ends apply
     * the same envelope). **However**, the build-time mirror at
     * `extractApkCertificateSha256` (build.gradle.kts:418-439) calls
     * `CertificateFactory.generateCertificate(InputStream)` on the same
     * CERT.RSA bytes — which skips the PKCS#7 envelope. To match, we do
     * the same: feed the whole stream into CertificateFactory.
     */
    private fun skipAsn1Envelope(bytes: ByteArray): ByteArray? = try {
        val cf = java.security.cert.CertificateFactory.getInstance("X.509")
        cf.generateCertificate(java.io.ByteArrayInputStream(bytes)).encoded
    } catch (t: Throwable) { null }

    private fun ByteArray.toLowerHex(): String =
        joinToString(separator = "") { "%02x".format(it) }
}
```
